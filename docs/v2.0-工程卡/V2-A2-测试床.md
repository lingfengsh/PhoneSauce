```python
# -*- coding: utf-8 -*-
"""合成信号 fixtures 生成器。运行一次生成 tests/fixtures/ 下的 wav 与锚点。

所有信号 Fs=16000 Hz、16bit PCM 单声道，参数均为解析已知：
  sine100   : 100 Hz 正弦，幅度 0.5，1.0 s（16000 样本=100 个整周期 → RMS 精确）
  harm5     : 基频 200 Hz，5 谐波，幅度 [0.5,0.25,0.125,0.0625,0.03125]，0.5 s
              （幅度整体 ×0.5 相对早期版本，使解析峰值 0.687 < 1.0，避免 16bit 硬截幅；
               见 V2-A2 卡 §11.5 缺陷记录）
  noise10   : 白噪声 sigma=0.1，seed=42（np.random.RandomState(42).normal），1.0 s
  silence   : 全零 0.5 s
  chirpF0   : F0 线性 100→300 Hz 的正弦扫频，幅度 0.4，2.0 s
锚点推导见 docs/v2.0-工程卡/V2-A2-测试床.md §6。禁止改动任何参数。
"""
import json
import math
import os
import wave

import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
FIX = os.path.join(HERE, "fixtures")
FS = 16000


def _write_wav(name, x):
    x16 = np.clip(np.round(x * 32767.0), -32768, 32767).astype(np.int16)
    with wave.open(os.path.join(FIX, name), "wb") as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(FS)
        w.writeframes(x16.tobytes())


def main():
    os.makedirs(FIX, exist_ok=True)
    os.makedirs(os.path.join(FIX, "expected"), exist_ok=True)
    os.makedirs(os.path.join(FIX, "snapshots"), exist_ok=True)
    rng = np.random.RandomState(42)  # noise10 与后续随机信号统一用此 seed

    t1 = np.arange(FS) / FS
    sine100 = 0.5 * np.sin(2 * np.pi * 100.0 * t1)
    _write_wav("sine100.wav", sine100)

    t2 = np.arange(FS // 2) / FS
    harm5 = np.zeros_like(t2)
    # 幅度整体 ×0.5（相对 v1）：解析峰值 0.687 < 1.0，16bit 不截幅。
    # 原 [1.0,0.5,0.25,0.125,0.0625] 峰值达 1.3745，被 np.clip 压平 27.5% 样本 → 能量锚点不可达。
    for k, amp in enumerate([0.5, 0.25, 0.125, 0.0625, 0.03125], start=1):
        harm5 += amp * np.sin(2 * np.pi * 200.0 * k * t2)
    _write_wav("harm5.wav", harm5)

    noise10 = 0.1 * rng.standard_normal(FS)
    _write_wav("noise10.wav", noise10)

    _write_wav("silence.wav", np.zeros(FS // 2))

    t3 = np.arange(2 * FS) / FS
    f_inst = 100.0 + (300.0 - 100.0) * t3 / 2.0          # 瞬时频率 100→300 Hz 线性
    chirpF0 = 0.4 * np.sin(2 * np.pi * np.cumsum(f_inst) / FS)  # 相位=频率积分
    _write_wav("chirpF0.wav", chirpF0)

    anchor = {
        "fs": FS,
        "sine100": {"rms": 0.5 / math.sqrt(2.0), "f0_hz": 100.0, "dur_s": 1.0},
        "harm5": {
            "f0_hz": 200.0,
            "dur_s": 0.5,
            "harmonics_hz": [200.0, 400.0, 600.0, 800.0, 1000.0],
            "amplitudes": [0.5, 0.25, 0.125, 0.0625, 0.03125],
            "peak": 0.6872753980252196,
            "energy_5periods_400samp": 66.6015625,
        },
        "noise10": {"sigma": 0.1, "seed": 42, "dur_s": 1.0},
        "silence": {"rms": 0.0, "dur_s": 0.5},
        "chirpF0": {"f0_start_hz": 100.0, "f0_end_hz": 300.0, "amp": 0.4, "dur_s": 2.0},
    }
    with open(os.path.join(FIX, "expected", "anchor.json"), "w", encoding="utf-8") as f:
        json.dump(anchor, f, ensure_ascii=False, indent=1)
    print("fixtures OK ->", FIX)


if __name__ == "__main__":
    main()
```

## 3. `tests/_snapshot.py`（完整代码，原样照抄）

```python
# -*- coding: utf-8 -*-
"""回归快照 helper。

用法（算法卡测试模板）：
    from tests._snapshot import assert_snapshot, SNAP_DIR

    def test_energy_snapshot():
        got = {"col": [ ... float ... ]}        # 键=样本名，值=float 或 list[float]
        assert_snapshot("get_energy", got)

- 快照存在：逐键对比，容差 abs 1e-9（NaN==NaN 视为相等）。
- 快照不存在：skip 并打印生成指引（由验收员运行 save 端生成，施工者禁止自行 save）。
"""
import json
import math
import os

import numpy as np

HERE = os.path.dirname(os.path.abspath(__file__))
SNAP_DIR = os.path.join(HERE, "fixtures", "snapshots")
TOL = 1e-9


def _norm(x):
    if isinstance(x, float) and math.isnan(x):
        return "NaN"
    if isinstance(x, (np.floating,)):
        return "NaN" if np.isnan(x) else round(float(x), 12)
    if isinstance(x, (int, np.integer)):
        return int(x)
    if isinstance(x, list):
        return [_norm(v) for v in x]
    if isinstance(x, dict):
        return {k: _norm(v) for k, v in sorted(x.items())}
    return x


def _path(name):
    return os.path.join(SNAP_DIR, name + ".json")


def assert_snapshot(name, got):
    fp = _path(name)
    assert os.path.isfile(fp), (
        f"快照 {name}.json 不存在。这是正常现象（首次实现）。"
        f"请在报告中注明『待验收员生成快照』，禁止自行生成。"
    )
    with open(fp, encoding="utf-8") as f:
        exp = json.load(f)
    got_n, exp_n = _norm(got), _norm(exp)
    assert set(got_n) == set(exp_n), f"键不一致: got={sorted(got_n)} exp={sorted(exp_n)}"

    def walk(g, e, path):
        if isinstance(g, dict):
            for k in g:
                walk(g[k], e[k], f"{path}.{k}")
        elif isinstance(g, list):
            assert len(g) == len(e), f"{path}: 长度 {len(g)} != {len(e)}"
            for i, (a, b) in enumerate(zip(g, e)):
                walk(a, b, f"{path}[{i}]")
        else:
            if g == "NaN" or e == "NaN":
                assert g == e, f"{path}: NaN 不匹配 got={g} exp={e}"
            else:
                assert abs(float(g) - float(e)) <= TOL, f"{path}: {g} != {e} (tol {TOL})"

    walk(got_n, exp_n, name)


def save_snapshot(name, got):
    """仅验收员使用：把当前输出冻结为新快照。"""
    os.makedirs(SNAP_DIR, exist_ok=True)
    with open(_path(name), "w", encoding="utf-8") as f:
        json.dump(_norm(got), f, ensure_ascii=False, indent=1)
    print("saved", _path(name))
```

## 5. `tests/test_fixtures.py`（完整代码，原样照抄）

```python
# -*- coding: utf-8 -*-
"""fixtures 自检：生成器跑完后这些测试必须全绿。"""
import json
import os
import wave

import numpy as np

FIX = os.path.join(os.path.dirname(os.path.abspath(__file__)), "fixtures")


def _read(name):
    with wave.open(os.path.join(FIX, name), "rb") as w:
        assert w.getframerate() == 16000 and w.getnchannels() == 1 and w.getsampwidth() == 2
        x = np.frombuffer(w.readframes(w.getnframes()), dtype=np.int16).astype(np.float64) / 32767.0
    return x


def test_anchor_exists():
    with open(os.path.join(FIX, "expected", "anchor.json"), encoding="utf-8") as f:
        a = json.load(f)
    assert a["fs"] == 16000


def test_sine100_rms_exact():
    x = _read("sine100.wav")
    # 16bit 量化引入 ~1e-5 误差，锚点容差放宽到 1e-3
    assert abs(np.sqrt(np.mean(x ** 2)) - 0.5 / np.sqrt(2.0)) < 1e-3
    assert len(x) == 16000


def test_harm5_peak_at_200hz():
    x = _read("harm5.wav")
    spec = np.abs(np.fft.rfft(x))
    freqs = np.fft.rfftfreq(len(x), 1.0 / 16000.0)
    for f in [200, 400, 600, 800, 1000]:
        assert spec[np.argmin(np.abs(freqs - f))] > 0  # 各谐波峰存在
    i200 = np.argmin(np.abs(freqs - 200.0))
    assert np.argmax(spec) == i200  # 全谱最大峰在 200 Hz（基频幅度最大）


def test_silence_is_zero():
    x = _read("silence.wav")
    assert float(np.max(np.abs(x))) == 0.0


def test_chirp_f0_range():
    x = _read("chirpF0.wav")
    # 粗检：过零率推平均 F0 应在 150~250 Hz 之间
    zc = np.nonzero(np.diff(np.signbit(x)))[0]
    mean_f0 = len(zc) / 2.0 / (len(x) / 16000.0)
    assert 150.0 < mean_f0 < 250.0
```

## 6. 锚点推导（供验收员核对，不许改动）

- `sine100` RMS：峰 0.5 的正弦 RMS = 0.5/√2 ≈ 0.353553。16000 样本 = 恰 100 个整周期 → 无截断误差；16bit 量化误差 ~3e-5，测试容差 1e-3。
- `harm5` 各谐波 DFT 峰幅度 = 系数×N/2（N=8000），基频系数最大 → rFFT 最大峰必在 200 Hz。
  - **幅度 [0.5,0.25,0.125,0.0625,0.03125]**（2026-10-26 由 [1.0,…,0.0625] 整体 ×0.5，见 §11.5）。
  - 解析峰值 = 0.6873 < 1.0 → 16bit **不截幅**（原幅度峰值 1.3745 会被 `np.clip` 压平 27.5% 样本）。
  - 5 周期窗能量 = Σamp²/2 ×400 = 0.3330078125/2 ×400 = **66.6015625**（解析精确；量化后实测 66.5974，差 0.0042，`atol=0.5` 内）。
- `chirpF0` 相位 = 2π·∫f dt（cumsum/FS 实现），平均 F0 = 200 Hz，过零率粗检区间 [150,250]。
- 量化：`np.round(x*32767)` 对称写法（不用 32768），保证正负峰对称、复现稳定。
- **⚠ 生成器统一 `np.clip(±32767)`，故所有信号解析峰值必须 < 1.0**，否则被硬截幅、锚点不可达（§11.5）。当前各信号峰值：sine100=0.5、harm5=0.687、noise10≈0.45、silence=0、chirpF0=0.4，均安全。

### 11.5 缺陷记录与修正：harm5 16bit 硬截幅（2026-10-26，用户拍板方案 A）

**问题**（施工 B1 卡时发现）：
- 原 `harm5` 幅度 `[1.0,0.5,0.25,0.125,0.0625]` 叠加后**解析峰值 1.3745 > 1.0**。
- 生成器 `_write_wav` 统一 `np.clip(round(x*32767), -32768, 32767)` → **2200/8000 = 27.5% 样本被压平到 ±1.0**，信号被系统性破坏。
- 结果：卡 §6 期望能量 `266.40625`（解析正确）在 wav 上**不可达**，实测仅 **205.7425**。
- 影响面：卡 §4 `test_harm5_constant_f0`（`atol=0.5`）必失败；且 B1 之后所有引用 harm5 的卡（C1/C2/C3/C4/C5/D1/D2）锚点全部污染。

**根因**：§6 锚点用**未量化解析值**推导，而 wav 存**量化后**值；`harm5` 是 5 个 fixture 中唯一解析峰值 > 1.0 者（其余均 < 0.5）。

**修正（方案 A，用户 2026-10-26 拍板）**：
- `harm5` 五谐波幅度**整体 ×0.5** → `[0.5,0.25,0.125,0.0625,0.03125]`，解析峰值 **0.6873 < 1.0**，不再截幅（实测压平样本 = 0）。
- anchor 能量重算：`Σamp²/2 ×400 = 0.3330078125/2 ×400 = **66.6015625**`（量化后实测 66.5974）。
- 比例关系仍整齐（1 : ½ : ¼ : ⅛ : 1/16）。
- 同步改：本卡 §2 生成器代码、§5 幅度、§6 锚点、§4 测试（`test_harm5_constant_f0` 期望值 266.40625 → 66.6015625，`atol=0.5` 不变）、`anchor.json`。

**§4 测试修正**：`assert np.allclose(mid, 266.40625, atol=0.5)` → `assert np.allclose(mid, 66.6015625, atol=0.5)`。

**附带修正（缺陷 2，sine100 测试容差）**：`test_sine100_constant_f0` 原 `atol=1e-6` 对 16bit 量化后的 wav（实测 99.99473，偏离 5.3e-3）**必失败**。修正为 `atol=1e-2`（量化误差 ~5e-3，1e-2 有余量）。此为纯测试容差修正，解析值 100.0 本身正确。

**遗留提醒**：`harm5` 幅度已变，凡在 §6/各算法卡中以 `[1.0,0.5,…]` 写死的 harm5 期望值，**以本卡 §6 新值 66.6015625 与 §11.5 为准**；B1 起的算法卡测试代码须用新幅度。
